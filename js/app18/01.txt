  function loadYouTubeApi(){
    if(window.YT?.Player)return Promise.resolve(window.YT);
    if(ytApiPromise)return ytApiPromise;
    ytApiPromise=new Promise((resolve,reject)=>{
      let settled=false;
      const previous=window.onYouTubeIframeAPIReady;
      window.onYouTubeIframeAPIReady=()=>{
        try{if(typeof previous==='function')previous()}catch{}
        settled=true;
        resolve(window.YT);
      };
      if(!document.querySelector('script[data-sm-youtube-api]')){
        const tag=document.createElement('script');
        tag.src='https://www.youtube.com/iframe_api';tag.async=true;tag.dataset.smYoutubeApi='1';
        tag.onerror=()=>reject(new Error('youtube api unavailable'));
        document.head.appendChild(tag);
      }
      let tries=0;
      const timer=setInterval(()=>{
        if(window.YT?.Player){clearInterval(timer);if(!settled)resolve(window.YT)}
        else if(++tries>120){clearInterval(timer);reject(new Error('youtube api timeout'))}
      },100);
    });
    return ytApiPromise;
  }
