  function normalizeOriginal(raw){
    const arr=Array.isArray(raw)?raw:(Array.isArray(raw?.questions)?raw.questions:[]);
    const strip=v=>String(v??'').replace(/^\s*[أ-يA-Za-z]\s*[\)\.\-:]\s*/, '').trim();
    const letters=['أ','ب','ج','د','هـ','ه','و'];
    const out=[];
    for(const q of arr){
      if(!q||!q.question||!q.options)continue;
      let options=[],answer=null;
      if(Array.isArray(q.options)){
        options=q.options.map(String);
        if(typeof q.correctAnswer==='number')answer=options[q.correctAnswer]??null;
        else if(typeof q.correctAnswer==='string'){
          const a=q.correctAnswer.trim();
          answer=options.find(o=>o===a||strip(o)===a)||a;
        }else if(typeof q.answer==='number')answer=options[q.answer]??null;
        else if(typeof q.answer==='string'){
          const a=q.answer.trim(),li=letters.indexOf(a);
          answer=li>=0&&li<options.length?options[li]:(options.find(o=>o===a||strip(o)===a)||a);
        }
      }else if(typeof q.options==='object'){
        const keys=Object.keys(q.options);options=keys.map(k=>String(q.options[k]));
        if(typeof q.answer==='string'){
          const a=q.answer.trim();answer=q.options[a]!==undefined?String(q.options[a]):(options.find(o=>o===a)||a);
        }else if(typeof q.correctAnswer==='string'){
          const a=q.correctAnswer.trim();answer=options.find(o=>o===a||strip(o)===a)||a;
        }else if(typeof q.correctAnswer==='number')answer=options[q.correctAnswer]??null;
      }
      if(answer!=null&&options.length>=2)out.push({question:String(q.question),options,answer:String(answer)});
    }
    return out;
  }

  async function fileToAttachment(file){
    if(file.size>10*1024*1024)throw new Error('حجم الملف أكبر من 10MB');
    const name=file.name.toLowerCase();
    if(file.type.startsWith('image/')){
      const dataUrl=await new Promise((res,rej)=>{const r=new FileReader();r.onload=()=>res(r.result);r.onerror=rej;r.readAsDataURL(file)});
      return {type:'image',name:file.name,mime:file.type,dataUrl};
    }
    if(file.type.startsWith('text/')||/\.(txt|md|csv)$/.test(name)){
      return {type:'file',name:file.name,mime:file.type||'text/plain',text:(await file.text()).slice(0,50000)};
    }
    if(name.endsWith('.pdf')){
      if(!window.pdfjsLib)throw new Error('مكتبة PDF غير متاحة. تحقق من الاتصال بالإنترنت.');
      window.pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
      const buf=await file.arrayBuffer();
      const doc=await window.pdfjsLib.getDocument({data:buf}).promise;
      let out='';
      for(let i=1;i<=doc.numPages;i++){
        const pg=await doc.getPage(i), content=await pg.getTextContent();
        out+=content.items.map(it=>it.str).join(' ')+'\n\n';
        if(out.length>50000)break;
      }
      return {type:'file',name:file.name,mime:'application/pdf',text:out.slice(0,50000)};
    }
    if(name.endsWith('.docx')){
      if(!window.mammoth)throw new Error('مكتبة DOCX غير متاحة. تحقق من الاتصال بالإنترنت.');
      const r=await window.mammoth.extractRawText({arrayBuffer:await file.arrayBuffer()});
      return {type:'file',name:file.name,mime:'application/vnd.openxmlformats-officedocument.wordprocessingml.document',text:String(r.value||'').slice(0,50000)};
    }
    if(/\.(xlsx|xls)$/.test(name)){
      if(!window.XLSX)throw new Error('مكتبة Excel غير متاحة. تحقق من الاتصال بالإنترنت.');
      const wb=window.XLSX.read(await file.arrayBuffer(),{type:'array'});let out='';
      for(const sn of wb.SheetNames){out+=`### ${sn}\n`+window.XLSX.utils.sheet_to_csv(wb.Sheets[sn])+'\n\n';if(out.length>50000)break;}
      return {type:'file',name:file.name,mime:file.type||'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',text:out.slice(0,50000)};
    }
    throw new Error('الأنواع المدعومة: صور، PDF، DOCX، XLSX/XLS، TXT، CSV، MD');
  }
