  const ytId=u=>{
    try{
      if(!u)return '';
      const x=new URL(String(u).trim());
      const host=x.hostname.replace(/^www\./,'').replace(/^m\./,'');
      let id='';
      if(host==='youtu.be') id=x.pathname.split('/').filter(Boolean)[0]||'';
      else if(host.endsWith('youtube.com')){
        if(x.searchParams.get('v')) id=x.searchParams.get('v')||'';
        else{
          const parts=x.pathname.split('/').filter(Boolean);
          const ix=parts.findIndex(p=>['embed','shorts','live'].includes(p));
          if(ix>=0) id=parts[ix+1]||'';
        }
      }
      id=id.split(/[?&#]/)[0];
      return /^[A-Za-z0-9_-]{6,20}$/.test(id)?id:'';
    }catch{return ''}
  };
